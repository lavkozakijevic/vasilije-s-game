function dirtPix(x,y,s){const h=hash(x,y,s);return h<.07?AMBER:h>.93?PLUM:BARK;}
function pebbles(g,s,n=3){const r=rng(s*7+3);for(let i=0;i<n;i++){const px=3+Math.floor(r()*25),py=9+Math.floor(r()*18);if(g.get(px,py)<0||g.get(px+1,py+1)<0)continue;g.set(px,py,STONE);g.set(px+1,py,SLATE);g.set(px,py+1,PLUM);g.set(px+1,py+1,PLUM);}}
function terrain(o,s=1){const g=new G(32,32),x0=o.l?1:0,x1=o.r?30:31,y0=o.t?1:0,y1=o.b?30:31,R=3;
const cut=(x,y,cx,cy)=>Math.hypot(x-cx,y-cy)>R+.3;
for(let y=y0;y<=y1;y++)for(let x=x0;x<=x1;x++){
if(o.t&&o.l&&x<x0+R&&y<y0+R&&cut(x,y,x0+R,y0+R))continue;if(o.t&&o.r&&x>x1-R&&y<y0+R&&cut(x,y,x1-R,y0+R))continue;
if(o.b&&o.l&&x<x0+R&&y>y1-R&&cut(x,y,x0+R,y1-R))continue;if(o.b&&o.r&&x>x1-R&&y>y1-R&&cut(x,y,x1-R,y1-R))continue;
let c=dirtPix(x,y,s);const gy=y-y0,gh=4+Math.floor(hash(x,0,5)*2);const grass=o.t&&gy<=gh;
if(o.t){if(gy===0)c=LEAF;else if(gy<=gh)c=MOSS;else if(gy===gh+1||(gy===gh+2&&hash(x,1,9)<.4))c=PINE;}
if(o.t&&o.l&&x<=x0+1&&gy<=gh+3)c=x===x0&&gy<=gh?LEAF:MOSS;
if(o.t&&o.r&&x>=x1-1&&gy<=gh+3)c=MOSS;
if(!grass){if(o.l&&x===x0&&c===BARK)c=AMBER;if(o.r&&x===x1)c=PLUM;if(o.r&&x===x1-1&&(x+y)%2===0)c=PLUM;if(o.b&&y===y1)c=PLUM;if(o.b&&y===y1-1&&(x+y)%2===0)c=PLUM;}
g.set(x,y,c);}

return g;}
